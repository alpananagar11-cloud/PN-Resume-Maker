    <script>

        // State Management for Dynamic Sections

        let experiences = [];

        let projects = [];

        let education = [];

        let currentZoom = 1;

        // Mobile Tab Switching

        function switchTab(tab) {

            const editorPane = document.getElementById('editor-pane');

            const previewPane = document.getElementById('preview-pane');

            const tabEditor = document.getElementById('tab-editor');

            const tabPreview = document.getElementById('tab-preview');

            if (tab === 'editor') {

                editorPane.classList.remove('hidden');

                previewPane.classList.add('hidden');

                tabEditor.className = "px-3 py-1.5 text-xs font-medium rounded-lg bg-indigo-600 text-white transition-all";

                tabPreview.className = "px-3 py-1.5 text-xs font-medium rounded-lg text-slate-400 hover:text-white transition-all";

            } else {

                editorPane.classList.add('hidden');

                previewPane.classList.remove('hidden');

                tabPreview.className = "px-3 py-1.5 text-xs font-medium rounded-lg bg-indigo-600 text-white transition-all";

                tabEditor.className = "px-3 py-1.5 text-xs font-medium rounded-lg text-slate-400 hover:text-white transition-all";

            }

        }

        // Toggle Category & Projects

        function toggleCategory() {

            const category = document.getElementById('jobCategory').value;

            const projectsSection = document.getElementById('projectsSection');

            const prevProjectsSection = document.getElementById('prevProjectsSection');

            if (category === 'it') {

                projectsSection.classList.remove('hidden');

                prevProjectsSection.classList.remove('hidden');

            } else {

                projectsSection.classList.add('hidden');

                prevProjectsSection.classList.add('hidden');

            }

            updatePreview();

        }

        // Zoom Controls

        function zoomIn() {

            if (currentZoom < 1.4) {

                currentZoom += 0.1;

                applyZoom();

            }

        }

        function zoomOut() {

            if (currentZoom > 0.6) {

                currentZoom -= 0.1;

                applyZoom();

            }

        }

        function resetZoom() {

            currentZoom = 1;

            applyZoom();

        }

        function applyZoom() {

            document.getElementById('resumeCanvas').style.transform = `scale(${currentZoom})`;

            document.getElementById('zoomLevel').innerText = `${Math.round(currentZoom * 100)}%`;

        }

        // Dynamic Experience Handler

        function addExperience(data = { company: '', role: '', period: '', description: '' }) {

            const id = Date.now() + Math.random();

            experiences.push({ id, ...data });

            renderExperienceEditor();

            updatePreview();

        }

        function removeExperience(id) {

            experiences = experiences.filter(exp => exp.id !== id);

            renderExperienceEditor();

            updatePreview();

        }

        function renderExperienceEditor() {

            const container = document.getElementById('experienceContainer');

            container.innerHTML = '';

            experiences.forEach((exp, index) => {

                const div = document.createElement('div');

                div.className = "bg-slate-900/60 p-3 rounded-xl border border-slate-700 space-y-2 relative";

                div.innerHTML = `

                    <div class="flex justify-between items-center">

                        <span class="text-xs text-indigo-400 font-medium">Experience #${index + 1}</span>

                        <button onclick="removeExperience(${exp.id})" class="text-slate-400 hover:text-rose-400 text-xs"><i class="fa-solid fa-trash"></i></button>

                    </div>

                    <div class="grid grid-cols-1 sm:grid-cols-2 gap-2">

                        <input type="text" placeholder="Company Name" value="${exp.company}" oninput="updateExpData(${exp.id}, 'company', this.value)" class="bg-slate-900 border border-slate-700 rounded-lg px-2.5 py-1.5 text-xs">

                        <input type="text" placeholder="Role / Position" value="${exp.role}" oninput="updateExpData(${exp.id}, 'role', this.value)" class="bg-slate-900 border border-slate-700 rounded-lg px-2.5 py-1.5 text-xs">

                    </div>

                    <input type="text" placeholder="Period (e.g. 2022 - Present)" value="${exp.period}" oninput="updateExpData(${exp.id}, 'period', this.value)" class="w-full bg-slate-900 border border-slate-700 rounded-lg px-2.5 py-1.5 text-xs">
